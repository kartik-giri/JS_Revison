bun add @prisma/client @prisma/adapter-pg dotenv pg
bunx --bun prisma init --output ../app/generated/prisma

3. Add dotenv to prisma.config.ts -> import "dotenv/config"; 
4. add models/schema in schema.prims file
5. bunx prisma migrate dev --name init

6. Then gernate prisma client from schema.
